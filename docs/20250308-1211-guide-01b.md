## Core Type System Implementation

### Base Type Class Implementation

The foundational `Type` class should be implemented with strong type hints and a clean interface:

```python
# pyvider_cty/base.py
from abc import ABC, abstractmethod
from typing import Any, Generic, Optional, TypeVar, cast, TYPE_CHECKING

if TYPE_CHECKING:
    from .value import Value

T = TypeVar("T")

class Type(ABC, Generic[T]):
    """Base class for all CTY types."""
    
    @abstractmethod
    def validate(self, value: Any) -> T:
        """
        Validate and coerce a value to this type.
        
        Args:
            value: The value to validate
            
        Returns:
            The validated value, possibly coerced to the expected type
            
        Raises:
            TypeError: If the value cannot be validated as this type
        """
        pass

    @abstractmethod
    def equal(self, other: "Type") -> bool:
        """
        Check if this type is exactly equal to another type.
        
        Args:
            other: The type to compare with
            
        Returns:
            True if the types are exactly equal, False otherwise
        """
        pass

    @abstractmethod
    def usable_as(self, other: "Type") -> bool:
        """
        Check if this type can be used where the other type is expected.
        
        Args:
            other: The type to check compatibility with
            
        Returns:
            True if this type can be used where the other is expected, False otherwise
        """
        pass
        
    def __eq__(self, other: object) -> bool:
        """Implement equality operator."""
        if not isinstance(other, Type):
            return False
        return self.equal(other)
```

### Primitive Types Implementation

Each primitive type should follow this pattern for consistency:

```python
# pyvider_cty/primitives/string.py
from typing import Any, Optional, TYPE_CHECKING

from ..base import Type
from ..exceptions import ValidationError

if TYPE_CHECKING:
    from ..unknown import UnknownValue

class CtyString(Type[str]):
    """String type for Terraform values."""
    
    def validate(self, value: Any) -> str:
        """Validate value as string."""
        from ..unknown import UnknownValue
        
        # Handle special values
        if isinstance(value, UnknownValue):
            return value
            
        if value is None:
            return None
            
        # Type conversion for non-strings
        if not isinstance(value, str):
            try:
                # Auto-convert numbers and booleans to strings
                if isinstance(value, (int, float, bool)):
                    return str(value)
                raise TypeError(f"Expected string, got {type(value).__name__}")
            except Exception as e:
                raise ValidationError(f"Invalid string value: {e}")
                
        return value
    
    def equal(self, other: Type) -> bool:
        """Check equality with another type."""
        return isinstance(other, CtyString)
        
    def usable_as(self, other: Type) -> bool:
        """Check if this type can be used as another type."""
        return isinstance(other, CtyString)
        
    def __repr__(self) -> str:
        return "String"

# Factory function for the public API
def String() -> CtyString:
    """Create a string type."""
    return CtyString()
```

Similar implementations should be created for `Number` and `Bool`, with appropriate type conversion logic.

### Collection Types Implementation

Collection types require handling of element types:

```python
# pyvider_cty/collections/list.py
from typing import Any, Generic, List as PyList, TypeVar, TYPE_CHECKING

from ..base import Type
from ..exceptions import ValidationError

if TYPE_CHECKING:
    from ..unknown import UnknownValue

T = TypeVar("T")

class CtyList(Type[PyList[T]], Generic[T]):
    """List type for ordered sequences of same-typed values."""
    
    def __init__(self, element_type: Type[T]):
        """
        Initialize a list type.
        
        Args:
            element_type: Type of elements in the list
        """
        self.element_type = element_type
        
    def validate(self, value: Any) -> PyList[T]:
        """Validate as list of elements matching the element type."""
        from ..unknown import UnknownValue
        
        # Handle special values
        if isinstance(value, UnknownValue):
            return value
            
        if value is None:
            return None
            
        # Type validation
        if not isinstance(value, (list, tuple)):
            raise ValidationError(f"Expected list, got {type(value).__name__}")
            
        # Validate each element
        validated_list = []
        for i, item in enumerate(value):
            try:
                validated_item = self.element_type.validate(item)
                validated_list.append(validated_item)
            except ValidationError as e:
                # Enhance error with index information
                raise ValidationError(f"Invalid element at index {i}: {e}")
                
        return validated_list
        
    def equal(self, other: Type) -> bool:
        """Check equality with another type."""
        if not isinstance(other, CtyList):
            return False
        return self.element_type.equal(other.element_type)
        
    def usable_as(self, other: Type) -> bool:
        """Check if this type can be used as another type."""
        if not isinstance(other, CtyList):
            return False
        return self.element_type.usable_as(other.element_type)
        
    def __repr__(self) -> str:
        return f"List({self.element_type})"

# Factory function for the public API
def List(element_type: Type[T]) -> CtyList[T]:
    """Create a list type with the given element type."""
    return CtyList(element_type)
```

Similar patterns apply to `Map` and `Set` types.

### Structural Types Implementation

Structural types like `Object` and `Tuple` require more complex validation:

```python
# pyvider_cty/structural/object.py
from typing import Any, Dict as PyDict, Set as PySet, Optional, TypeVar, TYPE_CHECKING

from ..base import Type
from ..exceptions import ValidationError

if TYPE_CHECKING:
    from ..unknown import UnknownValue

class CtyObject(Type[PyDict[str, Any]]):
    """Object type with an attribute schema."""
    
    def __init__(self, attribute_types: PyDict[str, Type], 
                 optional_attributes: Optional[PySet[str]] = None):
        """
        Initialize an object type.
        
        Args:
            attribute_types: Mapping of attribute names to their types
            optional_attributes: Set of attribute names that are optional
        """
        self.attribute_types = attribute_types
        self.optional_attributes = optional_attributes or set()
        
    def validate(self, value: Any) -> PyDict[str, Any]:
        """Validate as an object with typed attributes."""
        from ..unknown import UnknownValue
        
        # Handle special values
        if isinstance(value, UnknownValue):
            return value
            
        if value is None:
            return None
            
        # Type validation
        if not isinstance(value, dict):
            raise ValidationError(f"Expected object (dict), got {type(value).__name__}")
            
        # Track missing required attributes
        missing_attrs = set(self.attribute_types.keys()) - set(value.keys())
        required_missing = missing_attrs - self.optional_attributes
        if required_missing:
            raise ValidationError(f"Missing required attributes: {', '.join(required_missing)}")
            
        # Check for unexpected attributes
        unexpected = set(value.keys()) - set(self.attribute_types.keys())
        if unexpected:
            raise ValidationError(f"Unexpected attributes: {', '.join(unexpected)}")
            
        # Validate each attribute
        validated_obj = {}
        for name, attr_value in value.items():
            attr_type = self.attribute_types.get(name)
            if attr_type:
                try:
                    validated_obj[name] = attr_type.validate(attr_value)
                except ValidationError as e:
                    # Enhance error with attribute name
                    raise ValidationError(f"Invalid value for attribute '{name}': {e}")
                    
        return validated_obj
        
    def get_attribute_type(self, name: str) -> Type:
        """Get the type of an attribute."""
        if name not in self.attribute_types:
            raise ValidationError(f"Object does not have attribute '{name}'")
        return self.attribute_types[name]
        
    def equal(self, other: Type) -> bool:
        """Check equality with another type."""
        if not isinstance(other, CtyObject):
            return False
            
        if set(self.attribute_types.keys()) != set(other.attribute_types.keys()):
            return False
            
        return all(
            self.attribute_types[name].equal(other.attribute_types[name])
            for name in self.attribute_types
        )
        
    def usable_as(self, other: Type) -> bool:
        """Check if this type can be used as another type."""
        if not isinstance(other, CtyObject):
            return False
            
        # All required attributes in the other type must be present and compatible
        other_required = set(other.attribute_types.keys()) - other.optional_attributes
        for name in other_required:
            if name not in self.attribute_types:
                return False
            if not self.attribute_types[name].usable_as(other.attribute_types[name]):
                return False
                
        return True
        
    def __repr__(self) -> str:
        attrs = ", ".join(f"{name}: {type_}" for name, type_ in self.attribute_types.items())
        return f"Object({{{attrs}}})"

# Factory function for the public API
def Object(attribute_types: PyDict[str, Type], 
           optional_attributes: Optional[PySet[str]] = None) -> CtyObject:
    """Create an object type with the given attribute types."""
    return CtyObject(attribute_types, optional_attributes)
```

## Value System Implementation

### Value Class Implementation

The core `Value` class should implement immutability and type checking:

```python
# pyvider_cty/value.py
from typing import Any, Generic, List as PyList, Dict as PyDict, TypeVar, TYPE_CHECKING, cast

from .base import Type
from .exceptions import OperationError

if TYPE_CHECKING:
    from .unknown import UnknownValue

T = TypeVar("T")

class Value(Generic[T]):
    """Immutable representation of a Terraform-compatible value with type information."""
    
    def __init__(self, raw_value: Any, type_def: Type[T]):
        """
        Initialize a value.
        
        Args:
            raw_value: The raw Python value
            type_def: The type of this value
        """
        self._raw_value = raw_value
        self._type = type_def
        
    @property
    def type(self) -> Type[T]:
        """The type of this value."""
        return self._type
        
    @property
    def raw_value(self) -> Any:
        """The raw Python value."""
        return self._raw_value
        
    def is_known(self) -> bool:
        """Whether this value is known (not unknown)."""
        from .unknown import UnknownValue
        return not isinstance(self._raw_value, UnknownValue)
        
    def is_null(self) -> bool:
        """Whether this value is null."""
        return self._raw_value is None
        
    def as_string(self) -> str:
        """Get the value as a string."""
        if not self.is_known():
            raise OperationError("Cannot convert unknown value to string")
            
        if self.is_null():
            raise OperationError("Cannot convert null value to string")
            
        return cast(str, self._raw_value)
        
    def as_number(self) -> float:
        """Get the value as a number."""
        if not self.is_known():
            raise OperationError("Cannot convert unknown value to number")
            
        if self.is_null():
            raise OperationError("Cannot convert null value to number")
            
        return cast(float, self._raw_value)
        
    def as_bool(self) -> bool:
        """Get the value as a boolean."""
        if not self.is_known():
            raise OperationError("Cannot convert unknown value to boolean")
            
        if self.is_null():
            raise OperationError("Cannot convert null value to boolean")
            
        return cast(bool, self._raw_value)
        
    def as_value_slice(self) -> PyList["Value"]:
        """Get the value as a list of values."""
        if not self.is_known():
            raise OperationError("Cannot convert unknown value to value slice")
            
        if self.is_null():
            raise OperationError("Cannot convert null value to value slice")
            
        # Assume this is a list type
        from .collections.list import CtyList
        if not isinstance(self._type, CtyList):
            raise OperationError(f"Cannot convert {self._type} to value slice")
            
        element_type = self._type.element_type
        return [Value(item, element_type) for item in cast(PyList, self._raw_value)]
        
    def as_value_map(self) -> PyDict[str, "Value"]:
        """Get the value as a map of values."""
        if not self.is_known():
            raise OperationError("Cannot convert unknown value to value map")
            
        if self.is_null():
            raise OperationError("Cannot convert null value to value map")
            
        # Handle different map-like types
        from .collections.map import CtyMap
        from .structural.object import CtyObject
        
        if isinstance(self._type, CtyMap):
            element_type = self._type.element_type
            return {k: Value(v, element_type) for k, v in cast(PyDict, self._raw_value).items()}
            
        if isinstance(self._type, CtyObject):
            return {
                k: Value(v, self._type.get_attribute_type(k)) 
                for k, v in cast(PyDict, self._raw_value).items()
            }
            
        raise OperationError(f"Cannot convert {self._type} to value map")
        
    def get_attr(self, name: str) -> "Value":
        """Get an attribute from an object value."""
        if not self.is_known():
            # Unknown values propagate
            from .unknown import UnknownVal
            from .structural.object import CtyObject
            if not isinstance(self._type, CtyObject):
                raise OperationError(f"Cannot get attribute from {self._type}")
            return UnknownVal(self._type.get_attribute_type(name))
            
        if self.is_null():
            raise OperationError("Cannot get attribute from null value")
            
        # Handle object types
        from .structural.object import CtyObject
        if not isinstance(self._type, CtyObject):
            raise OperationError(f"Cannot get attribute from {self._type}")
            
        attr_type = self._type.get_attribute_type(name)
        attr_value = cast(PyDict, self._raw_value).get(name)
        
        return Value(attr_value, attr_type)
        
    def __eq__(self, other: object) -> bool:
        """Implement equality operator."""
        if not isinstance(other, Value):
            return False
            
        # Unknown values are only equal to themselves
        if not self.is_known() or not other.is_known():
            return self is other
            
        # Null values of the same type are equal
        if self.is_null() and other.is_null():
            return self._type.equal(other._type)
            
        # Otherwise, compare types and raw values
        return (
            self._type.equal(other._type) and 
            self._raw_value == other._raw_value
        )
        
    def __repr__(self) -> str:
        if not self.is_known():
            return f"UnknownVal({self._type})"
            
        if self.is_null():
            return f"NullVal({self._type})"
            
        return f"Value({self._raw_value!r}, {self._type})"
```

### Unknown Value Implementation

The unknown value system requires special handling:

```python
# pyvider_cty/unknown.py
from typing import Any, Callable, List, Optional, TYPE_CHECKING

if TYPE_CHECKING:
    from .base import Type
    from .value import Value

class UnknownValue:
    """Represents an unknown value in Terraform."""
    
    def __init__(self, refinements: Optional[List[Callable[[Any], bool]]] = None):
        """
        Initialize an unknown value, optionally with refinements.
        
        Args:
            refinements: List of refinement functions that constrain what this unknown might be
        """
        self.refinements = refinements or []
        
    def apply_refinements(self, value: Any) -> bool:
        """
        Apply all refinements to a value.
        
        Args:
            value: The value to check against refinements
            
        Returns:
            True if the value passes all refinements, False otherwise
        """
        return all(refine(value) for refine in self.refinements)
        
    def __repr__(self) -> str:
        return "UnknownValue"

def UnknownVal(type_def: "Type") -> "Value":
    """
    Create an unknown value of the specified type.
    
    Args:
        type_def: The type of the unknown value
        
    Returns:
        An unknown value of the specified type
    """
    from .value import Value
    return Value(UnknownValue(), type_def)

class RefinementBuilder:
    """Builder for refining unknown values with constraints."""
    
    def __init__(self, value: "Value"):
        """
        Initialize a refinement builder.
        
        Args:
            value: The unknown value to refine
        """
        from .value import Value
        
        if not isinstance(value, Value):
            raise TypeError("Can only refine Value objects")
            
        if value.is_known():
            raise ValueError("Can only refine unknown values")
            
        self.value = value
        self.constraints: List[Callable[[Any], bool]] = []
        
        # Extract existing refinements if present
        raw_value = value.raw_value
        if isinstance(raw_value, UnknownValue) and raw_value.refinements:
            self.constraints.extend(raw_value.refinements)
        
    def not_null(self) -> "RefinementBuilder":
        """Mark that this unknown value cannot be null."""
        self.constraints.append(lambda v: v is not None)
        return self
        
    def string_prefix(self, prefix: str) -> "RefinementBuilder":
        """Specify that this unknown string value has a known prefix."""
        self.constraints.append(lambda v: isinstance(v, str) and v.startswith(prefix))
        return self
        
    def number_min(self, min_value: float) -> "RefinementBuilder":
        """Specify that this unknown number value has a minimum value."""
        self.constraints.append(lambda v: isinstance(v, (int, float)) and v >= min_value)
        return self
        
    def number_max(self, max_value: float) -> "RefinementBuilder":
        """Specify that this unknown number value has a maximum value."""
        self.constraints.append(lambda v: isinstance(v, (int, float)) and v <= max_value)
        return self
        
    def build(self) -> "Value":
        """Build the refined value."""
        from .value import Value
        
        # Create a new unknown value with the refined constraints
        refined_value = UnknownValue(self.constraints)
        return Value(refined_value, self.value.type)
```

### Path Implementation

The path system allows referencing parts of nested values:

```python
# pyvider_cty/path.py
from abc import ABC, abstractmethod
from typing import List as PyList, Optional, TYPE_CHECKING

if TYPE_CHECKING:
    from .value import Value

class PathStep(ABC):
    """Abstract base class for path steps."""
    
    @abstractmethod
    def apply(self, value: "Value") -> "Value":
        """Apply this step to a value."""
        pass
        
    @abstractmethod
    def __str__(self) -> str:

class GetAttrStep(PathStep):

    def apply(self, value: "Value") -> "Value":
        return value.get_attr(self.name)
        
    def __str__(self) -> str:
        return f".{self.name}"
        
class IndexStep(PathStep):
    """Step that indexes into a collection."""
    
    def __init__(self, key: Any):
        self.key = key
        
    def apply(self, value: "Value") -> "Value":
        # Implementation for indexing
        # ...
        
    def __str__(self) -> str:
        if isinstance(self.key, int):
            return f"[{self.key}]"
        else:
            return f"[{self.key!r}]"

class Path:
    """Represents a path to a value within a nested structure."""
    
    @staticmethod
    def root() -> "Path":
        """Create a root path."""
        return Path([])
        
    def __init__(self, steps: PyList[PathStep]):
        self.steps = steps
        
    def get_attr(self, name: str) -> "Path":
        """Add an attribute access step to this path."""
        return Path(self.steps + [GetAttrStep(name)])
        
    def index(self, key: Any) -> "Path":
        """Add an index access step to this path."""
        return Path(self.steps + [IndexStep(key)])
        
    def apply(self, value: "Value") -> "Value":
        """Apply this path to a value to extract a nested value."""
        result = value
        for step in self.steps:
            result = step.apply(result)
        return result
        
    def __str__(self) -> str:
        if not self.steps:
            return "$"
        return "$" + "".join(str(step) for step in self.steps)
```

## Serialization System Implementation

### JSON Encoding Implementation

```python
# pyvider_cty/encoding/json.py
import json
from typing import Any, Optional, TYPE_CHECKING

if TYPE_CHECKING:
    from ..base import Type
    from ..value import Value
    from ..unknown import UnknownValue

class JSONEncoder:
    """Encodes CTY values to JSON."""
    
    @staticmethod
    def encode(value: "Value") -> bytes:
        """
        Encode a value to JSON bytes.
        
        Args:
            value: The value to encode
            
        Returns:
            JSON-encoded bytes
        """
        # Handle unknown values
        if not value.is_known():
            # Special marker for unknown values
            return b'"__UNKNOWN__"'
            
        if value.is_null():
            return b'null'
            
        # Convert to Python representation
        python_value = JSONEncoder._to_python(value)
        
        # Encode to JSON
        return json.dumps(python_value).encode('utf-8')
    
    @staticmethod
    def _to_python(value: "Value") -> Any:
        """Convert a value to a Python representation suitable for JSON."""
        # Handle primitive types directly
        from ..primitives.string import CtyString
        from ..primitives.number import CtyNumber
        from ..primitives.bool import CtyBool
        
        if isinstance(value.type, (CtyString, CtyNumber, CtyBool)):
            return value.raw_value
            
        # Handle collection types
        from ..collections.list import CtyList
        from ..collections.map import CtyMap
        
        if isinstance(value.type, CtyList):
            return [JSONEncoder._to_python(v) for v in value.as_value_slice()]
            
        if isinstance(value.type, CtyMap):
            return {k: JSONEncoder._to_python(v) for k, v in value.as_value_map().items()}
            
        # Handle structural types
        from ..structural.object import CtyObject
        from ..structural.tuple import CtyTuple
        
        if isinstance(value.type, CtyObject):
            return {k: JSONEncoder._to_python(v) for k, v in value.as_value_map().items()}
            
        if isinstance(value.type, CtyTuple):
            # Similar to list handling
            # ...
            
        raise ValueError(f"Cannot encode {value.type} to JSON")

class JSONDecoder:
    """Decodes JSON to CTY values."""
    
    @staticmethod
    def decode(data: bytes, type_def: "Type") -> "Value":
        """
        Decode JSON bytes to a value.
        
        Args:
            data: The JSON-encoded bytes
            type_def: The expected type
            
        Returns:
            The decoded value
        """
        # Handle special markers
        if data == b'"__UNKNOWN__"':
            from ..unknown import UnknownVal
            return UnknownVal(type_def)
            
        if data == b'null':
            from ..value import Value
            return Value(None, type_def)
            
        # Parse JSON
        raw_value = json.loads(data.decode('utf-8'))
        
        # Validate against type
        from ..value import Value
        validated_value = type_def.validate(raw_value)
```

```python
from typing import Optional, TYPE_CHECKING

if TYPE_CHECKING:
    from ..base import Type
    from ..value import Value
    from pyvider.protocols.tfprotov6.protobuf import DynamicValue

class DynamicValueHandler:
    """Handles conversion between Python values and Terraform DynamicValue."""
    
    @staticmethod
    def encode(value: "Value") -> "DynamicValue":
        """
        Convert a Value to a DynamicValue protobuf message.
        
        Args:
            value: The value to encode
            
        Returns:
            A DynamicValue protobuf message
        """
        from pyvider.protocols.tfprotov6.protobuf import DynamicValue
        
        if not value.is_known():
            # Handle unknown values specially
            return DynamicValue(unknown=True)
            
        # For known values, use JSON encoding
        from .json import JSONEncoder
        json_data = JSONEncoder.encode(value)
        return DynamicValue(json=json_data)
            
    @staticmethod
    def decode(dynamic_value: "DynamicValue", expected_type: "Type") -> "Value":
        """
        Convert a DynamicValue protobuf message to a Value.
        
        Args:
            dynamic_value: The DynamicValue to decode
            expected_type: The expected type
            
        Returns:
            The decoded value
        """
        if dynamic_value.unknown:
            from ..unknown import UnknownVal
            return UnknownVal(expected_type)
            
        if dynamic_value.json:
            from .json import JSONDecoder
            return JSONDecoder.decode(dynamic_value.json, expected_type)
            
        if dynamic_value.msgpack:
            from .msgpack import MsgpackDecoder
            return MsgpackDecoder.decode(dynamic_value.msgpack, expected_type)
            
        # Empty DynamicValue is treated as null
        from ..value import Value
        return Value(None, expected_type)
```

```python
class FieldDescriptor:
    """Descriptor for schema fields."""
    
    def __init__(self, required: bool = False, 
                 computed: bool = False,
                 optional: bool = False,
                 sensitive: bool = False,
                 description: Optional[str] = None,
                 default: Any = None,
                 validators: Optional[List[Callable[[Any], None]]] = None):
        """
        Initialize a field descriptor.
        
        Args:
            required: Whether the field is required
            computed: Whether the field is computed by the provider
            optional: Whether the field is optional
            sensitive: Whether the field is sensitive
            description: Description of the field
            default: Default value for the field
            validators: List of validator functions
        """
        self._field_metadata = {
            'required': required,
            'computed': computed,
            'optional': optional,
            'sensitive': sensitive,
            'description': description,
            'default': default,
            'validators': validators or []
        }
        self._name = None
            
    def __get__(self, obj, objtype=None):
        if obj is None:
            return self
        return obj.__dict__.get(self._name, self._field_metadata.get('default'))
            
    def __set__(self, obj, value):
        obj.__dict__[self._name] = value
            
    def __set_name__(self, owner, name):
        self._name = name
        self._field_metadata['name'] = name
```

### Resource Decorator Implementation

```python
# pyvider/schema/decorators.py
from typing import Any, Callable, Dict, List, Optional, Type as PyType, get_type_hints

def resource(cls=None, *, name: Optional[str] = None):
    """Decorator for defining Terraform resource schemas."""
    def wrapper(cls):
        annotations = get_type_hints(cls)
        attributes = {}
        
        # Process field annotations and metadata
        for field_name, field_type in annotations.items():
            if field_name.startswith('_'):
                continue
                
            field_obj = getattr(cls, field_name, None)
            if not hasattr(field_obj, '_field_metadata'):
                raise ValueError(f"Field {field_name} missing metadata. Use the field() function.")
                
            metadata = field_obj._field_metadata
            
            # Store attribute configuration
            attributes[field_name] = {
                'type': field_type,
                'required': metadata.get('required', False),
                'computed': metadata.get('computed', False),
                'optional': metadata.get('optional', False),
                'sensitive': metadata.get('sensitive', False),
                'description': metadata.get('description', ''),
                'default': metadata.get('default'),
                'validators': metadata.get('validators', []),
            }
        
        # Store schema info on the class
        cls._schema_name = name or cls.__name__.lower()
        cls._schema_attributes = attributes
        
        # Add schema methods
        cls.to_proto = schema_to_proto.__get__(cls)
        
        return cls
        
    if cls is None:
        return wrapper
    return wrapper(cls)

def schema_to_proto(cls):
    """Convert schema to Protocol Buffer format."""
    from pyvider.protocols.tfprotov6.protobuf import Schema, StringKind
    
    attributes = []
    for name, attr_info in cls._schema_attributes.items():
        cty_type = attr_info['type']
        
        # Convert CTY type to protocol buffer type
        pb_type = _convert_type_to_schema_type(cty_type)
        
        attributes.append(
            Schema.Attribute(
                name=name,
                type=pb_type,
                description=attr_info.get('description', name),
                required=attr_info.get('required', False),
                computed=attr_info.get('computed', False),
                optional=attr_info.get('optional', False),
                sensitive=attr_info.get('sensitive', False),
                description_kind=StringKind.PLAIN
            )
        )
    
    return Schema(
        version=1,
        block=Schema.Block(
            attributes=attributes,
            description=cls.__doc__ or cls.__name__,
            description_kind=StringKind.PLAIN
        )
    )

def _convert_type_to_schema_type(cty_type):
    """Convert a CTY type to a Protocol Buffer schema type."""
    from pyvider_cty.primitives.string import CtyString
    from pyvider_cty.primitives.number import CtyNumber
    from pyvider_cty.primitives.bool import CtyBool
    
    # Map CTY types to Schema.Attribute type strings
    if isinstance(cty_type, CtyString):
        return b'"string"'
    elif isinstance(cty_type, CtyNumber):
        return b'"number"'
    elif isinstance(cty_type, CtyBool):
        return b'"bool"'
        
    # Handle collection and structural types
    from pyvider_cty.collections.list import CtyList
    from pyvider_cty.collections.map import CtyMap
    
    if isinstance(cty_type, CtyList):
        return b'"list"'
    elif isinstance(cty_type, CtyMap):
        return b'"map"'
        
    # Handle nested objects
    from pyvider_cty.structural.object import CtyObject
    
    if isinstance(cty_type, CtyObject):
        return b'"object"'
        
    # Default to dynamic for unknown types
    return b'"dynamic"'
```
