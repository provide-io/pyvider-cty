#### Base Type Class
```python
# pyvider.cty/base.py
from abc import ABC, abstractmethod
from typing import Any, Generic, TypeVar

T = TypeVar("T")

class Type(ABC, Generic[T]):
    """Base class for all Cty types."""
    
    @abstractmethod
    def validate(self, value: Any) -> T:
        """Validate and possibly convert a value to this type."""
        
    @abstractmethod
    def equal(self, other: "Type") -> bool:
        """Check if two types are equal."""
    
    @abstractmethod
    def usable_as(self, other: "Type") -> bool:
        """Check if this type can be used where the other is expected."""
```

#### Primitive Type Example
```python
# pyvider.cty/primitives/string.py
from typing import Any
from pyvider.cty.base import Type

class String(Type[str]):
    """String type for Cty."""
    
    def validate(self, value: Any) -> str:
        if value is None:
            return ""
        if not isinstance(value, str):
            raise TypeError(f"Expected string, got {type(value).__name__}")
        return value
    
    def equal(self, other: Type) -> bool:
        return isinstance(other, String)
    
    def usable_as(self, other: Type) -> bool:
        return isinstance(other, String)
```

### 3.3 Schema System With Strawberry-like Interface

#### Decorator-based Schema Definition
```python
# pyvider/schema/decorators.py
from typing import Optional, Type, get_type_hints
import inspect

def resource(cls=None, *, name: Optional[str] = None):
    """Decorator to define a Terraform resource schema."""
    def wrap(cls):
        annotations = get_type_hints(cls)
        attributes = {}
        
        # Process field annotations and field objects
        for field_name, field_type in annotations.items():
            if field_name.startswith('_'):
                continue
                
            # Get field metadata
            field_obj = getattr(cls, field_name, None)
            required = getattr(field_obj, 'required', False)
            computed = getattr(field_obj, 'computed', False)
            description = getattr(field_obj, 'description', None)
            
            # Store the attribute information
            attributes[field_name] = {
                'type': field_type,
                'required': required,
                'computed': computed,
                'description': description,
                # Additional metadata...
            }
        
        # Store schema information on the class
        resource_name = name or cls.__name__.lower()
        cls._schema_name = resource_name
        cls._schema_attributes = attributes
        cls._schema_description = cls.__doc__ or f"{resource_name} resource"
        
        # Add schema generation methods
        cls.to_proto = schema_to_proto.__get__(cls)
        
        return cls
    
    # Handle both @resource and @resource(name="example") forms
    if cls is None:
        return wrap
    return wrap(cls)

def schema_to_proto(cls):
    """Convert schema to Protocol Buffer format."""
    from pyvider.protocols.tfprotov6.protobuf import Schema, StringKind
    
    attributes = []
    for name, attr_info in cls._schema_attributes.items():
        cty_type = attr_info['type']
        pb_type = _convert_type_to_schema_type(cty_type)
        
        attributes.append(
            Schema.Attribute(
                name=name,
                type=pb_type,
                description=attr_info.get('description', name),
                required=attr_info.get('required', False),
                computed=attr_info.get('computed', False),
                optional=attr_info.get('optional', False),
                sensitive=attr_info.get('sensitive', False),
                description_kind=StringKind.PLAIN
            )
        )
    
    return Schema(
        version=1,
        block=Schema.Block(
            attributes=attributes,
            description=cls._schema_description,
            description_kind=StringKind.PLAIN
        )
    )
```
