# Using Terraform State Files for Cty Testing

You've identified an excellent approach for validating your Cty implementation. Yes, `terraform.tfstate` files are indeed serialized using Go-Cty's JSON encoding and can serve as valuable test fixtures.

## How Terraform State Relates to Cty

The `terraform.tfstate` file is a JSON document where the resource attributes section uses Cty's JSON encoding format:

```json
{
  "version": 4,
  "terraform_version": "1.5.0",
  "resources": [
    {
      "mode": "managed",
      "type": "example_resource",
      "name": "example",
      "instances": [
        {
          "schema_version": 0,
          "attributes": {
            // This section uses Cty JSON encoding
            "id": "example123",
            "name": "test-resource",
            "tags": {
              "environment": "development",
              "owner": "team-a"
            },
            "list_attribute": ["value1", "value2"],
            "nested_object": {
              "foo": "bar",
              "baz": 42
            }
          }
        }
      ]
    }
  ]
}
```

### 2. Build Test Harnesses

Create test cases that:
1. Parse Terraform state files
2. Extract the attribute sections
3. Decode them with your Cty implementation
4. Verify correct interpretation

```python
import json
import pytest
from pyvider.cty import String, Number, Bool, List, Map, Object
from pyvider.cty.encoding import json as cty_json

def load_state_attributes(state_file):
    """Extract attributes section from terraform.tfstate file."""
    with open(state_file, 'r') as f:
        state = json.load(f)
    
    # Navigate to the attributes section
    resource = state['resources'][0]
    instance = resource['instances'][0]
    return instance['attributes']

@pytest.fixture
def tf_state_attributes():
    return load_state_attributes('tests/fixtures/terraform.tfstate')

def test_decode_primitive_types(tf_state_attributes):
    """Test decoding primitive types from Terraform state."""
    triggers = tf_state_attributes['triggers']
    
    # Define the schema for string value
    string_schema = String()
    string_value = cty_json.unmarshal(json.dumps(triggers['string_value']).encode(), string_schema)
    assert string_value.as_string() == "hello world"
    
    # Number value
    number_schema = Number()
    number_value = cty_json.unmarshal(json.dumps(triggers['number_value']).encode(), number_schema)
    assert number_value.as_number() == 42
    
    # Boolean value
    bool_schema = Bool()
    bool_value = cty_json.unmarshal(json.dumps(triggers['bool_value']).encode(), bool_schema)
    assert bool_value.as_bool() is True

def test_round_trip_serialization(tf_state_attributes):
    """Test that we can round-trip serialize/deserialize values."""
    triggers = tf_state_attributes['triggers']
    
    # Parse the nested object
    nested_json = json.loads(triggers['nested_object'])
    
    # Define the schema
    nested_schema = Object({
        "level1": Object({
            "level2": Object({
                "level3": String(),
                "number": Number()
            }),
            "array": List(Number())
        })
    })
    
    # Decode
    decoded = cty_json.unmarshal(json.dumps(nested_json).encode(), nested_schema)
    
    # Re-encode
    encoded = cty_json.marshal(decoded)
    
    # Parse back to Python dict for comparison
    round_tripped = json.loads(encoded.decode())
    
    # Should match the original
    assert round_tripped == nested_json
```

```python
def test_unknown_values():
    """Test handling of unknown values in Cty."""
    from pyvider.cty import UnknownVal
    
    # Create an unknown string value
    unknown_string = UnknownVal(String())
    
    # Encode it
    encoded = cty_json.marshal(unknown_string)
    
    # Should encode as a special marker in Terraform
    # Exact format depends on Terraform's serialization of unknowns
    assert b'"__unknown__"' in encoded or b'"__UNKNOWN__"' in encoded

def test_null_values():
    """Test handling of null values."""
    from pyvider.cty import NullVal
    
    # Create null values of different types
    null_string = NullVal(String())
    null_number = NullVal(Number())
    
    # Encode them
    encoded_string = cty_json.marshal(null_string)
    encoded_number = cty_json.marshal(null_number)
    
    # Both should encode as JSON null
    assert encoded_string == b'null'
    assert encoded_number == b'null'
    
    # But they should retain their type information when decoded
    decoded_string = cty_json.unmarshal(b'null', String())
    assert decoded_string.type == String()
    assert decoded_string.is_null()
```

### 4. Terraform Command Line Integration Tests

For even more robust validation, you can test against live Terraform:

```python
import subprocess
import tempfile
import os

def test_terraform_compatibility():
    """Test that our Cty implementation works with actual Terraform."""
    # Create a Terraform configuration
    tf_config = """
    output "test_output" {
      value = {
        string_val = "hello"
        number_val = 42
        list_val = ["a", "b", "c"]
      }
    }
    """
    
    # Write to temporary file
    with tempfile.TemporaryDirectory() as tmpdir:
        with open(os.path.join(tmpdir, "main.tf"), "w") as f:
            f.write(tf_config)
        
        # Initialize and apply
        subprocess.run(["terraform", "init"], cwd=tmpdir, check=True)
        subprocess.run(["terraform", "apply", "-auto-approve"], cwd=tmpdir, check=True)
        
        # Get output as JSON
        result = subprocess.run(
            ["terraform", "output", "-json", "test_output"], 
            cwd=tmpdir, 
            check=True,
            capture_output=True,
            text=True
        )
        
        # Parse the output
        tf_output = json.loads(result.stdout)
        
        # Define our schema
        schema = Object({
            "string_val": String(),
            "number_val": Number(),
            "list_val": List(String())
        })
        
        # Parse with our Cty implementation
        parsed = cty_json.unmarshal(json.dumps(tf_output).encode(), schema)
        
        # Validate
        assert parsed.get_attr("string_val").as_string() == "hello"
        assert parsed.get_attr("number_val").as_number() == 42
        assert parsed.get_attr("list_val").as_value_slice()[0].as_string() == "a"
```
